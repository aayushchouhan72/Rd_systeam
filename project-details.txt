    rduser_id INTEGER NOT NULL,                -- FK to rdusers

    rd_total_amount NUMERIC(12,2) NOT NULL,    -- total RD value
    paid_till_amount NUMERIC(12,2) DEFAULT 0,  -- how much paid so far

    installment_amount NUMERIC(10,2) NOT NULL, -- monthly installment
    duration_months INTEGER NOT NULL,          -- RD duration

    rd_start_date DATE NOT NULL,               -- RD start date
    monthly_installment_day INTEGER NOT NULL,  -- 1–28

    is_completed BOOLEAN DEFAULT FALSE,        -- RD completed or not
    is_settled BOOLEAN DEFAULT FALSE,           -- payout done or not

    created_at TIMESTAMP DEFAULT NOW(),

    CONSTRAINT fk_rd_user
      FOREIGN KEY (rduser_id)
      REFERENCES rdusers(id)
      ON DELETE CASCADE,

    CONSTRAINT chk_installment_day
      CHECK (monthly_installment_day BETWEEN 1 AND 28),

    CONSTRAINT chk_duration
